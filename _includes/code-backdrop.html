/* the same idea, in C */
#include <stdio.h>

int add(int a, int b) {
    return a + b;
}

int main(void) {
    printf("%d\n", add(2, 3));
    return 0;
}
// the same idea, in C#
using System;

public static class Program
{
    static int Add(int a, int b) => a + b;

    public static void Main()
    {
        Console.WriteLine(Add(2, 3));
    }
}
-- and finally, T-SQL
CREATE FUNCTION dbo.Add (@a INT, @b INT)
RETURNS INT
AS
BEGIN
    RETURN @a + @b;
END;
GO

SELECT dbo.Add(2, 3) AS Result;
